```lua
--// Services
local Players = game:GetService("Players")

--// Joueur local
local LocalPlayer = Players.LocalPlayer


--// Remplace le Humanoid actuel par une copie propre
local function ReplaceHumanoid(character)

    -- Récupération du Humanoid
    local oldHumanoid = character:FindFirstChildOfClass("Humanoid")

    if not oldHumanoid then
        return
    end


    -- Sauvegarde des propriétés importantes
    local walkSpeed = oldHumanoid.WalkSpeed
    local jumpPower = oldHumanoid.JumpPower
    local jumpHeight = oldHumanoid.JumpHeight
    local health = oldHumanoid.Health
    local maxHealth = oldHumanoid.MaxHealth


    -- Caméra actuelle
    local camera = workspace.CurrentCamera


    -- Script d'animation du personnage
    local animateScript = character:FindFirstChild("Animate")


    -- Désactivation temporaire des animations
    if animateScript and animateScript:IsA("LocalScript") then
        animateScript.Disabled = true
    end


    -- Arrêt des animations actuellement jouées
    local animator = oldHumanoid:FindFirstChildOfClass("Animator")

    if animator then
        for _, animationTrack in next, animator:GetPlayingAnimationTracks() do
            animationTrack:Stop(0)
        end
    end


    -- Autorise le clonage du Humanoid
    oldHumanoid.Archivable = true


    -- Création d'une copie du Humanoid
    local newHumanoid = oldHumanoid:Clone()


    -- Suppression de l'ancien Animator
    -- Un nouvel Animator sera créé ensuite.
    for _, child in next, newHumanoid:GetChildren() do
        if child:IsA("Animator") then
            child:Destroy()
        end
    end


    -- Renommage temporaire de l'ancien Humanoid
    oldHumanoid.Name = "_OldHumanoid"

    -- Le nouveau prend le nom normal
    newHumanoid.Name = "Humanoid"

    -- Ajout du nouveau Humanoid au personnage
    newHumanoid.Parent = character


    -- Création d'un nouvel Animator
    local newAnimator = Instance.new("Animator")
    newAnimator.Parent = newHumanoid


    -- Restauration des propriétés
    newHumanoid.WalkSpeed = walkSpeed
    newHumanoid.JumpPower = jumpPower
    newHumanoid.JumpHeight = jumpHeight
    newHumanoid.MaxHealth = maxHealth
    newHumanoid.Health = math.min(health, maxHealth)


    -- La caméra doit suivre le nouveau Humanoid
    if camera then
        camera.CameraSubject = newHumanoid
    end


    -- Suppression de l'ancien Humanoid
    oldHumanoid:Destroy()


    -- Réactivation temporaire du script d'animation
    if animateScript and animateScript:IsA("LocalScript") then

        task.wait()

        animateScript.Disabled = false

        -- Petite réinitialisation supplémentaire des animations
        task.defer(function()

            if animateScript.Parent then
                animateScript.Disabled = true

                task.wait()

                animateScript.Disabled = false
            end

        end)
    end


    -- Force le nouveau Humanoid à revenir à l'état Running
    task.defer(function()

        if newHumanoid.Parent then
            newHumanoid:ChangeState(Enum.HumanoidStateType.Running)
        end

    end)


    return newHumanoid
end


--// Fonction appelée lorsqu'un personnage apparaît
local function SetupCharacter(character)

    -- Attend que le Humanoid existe
    character:WaitForChild("Humanoid")

    -- Petit délai comme dans le script original
    task.wait()

    -- Remplacement du Humanoid
    ReplaceHumanoid(character)
end


--// Si le personnage existe déjà
if LocalPlayer.Character then
    task.spawn(SetupCharacter, LocalPlayer.Character)
end


--// Lorsque le joueur respawn
LocalPlayer.CharacterAdded:Connect(function(character)

    task.spawn(SetupCharacter, character)

end)
```
